cmake_minimum_required(VERSION 3.16)
project(viettelex_engine CXX)

# Pure C++17 (compiles as C++20 too). No platform headers, no exceptions/RTTI needed.
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(VTX_BUILD_TESTS "Build golden/unit tests and benchmark" ON)

add_library(viettelex_engine STATIC
  src/telex_engine.cpp
  src/syllable_validator.cpp
  src/vtx_engine.cpp)
target_include_directories(viettelex_engine PUBLIC include PRIVATE src)
if(MSVC)
  target_compile_options(viettelex_engine PRIVATE /W4 /permissive- /utf-8 /GR- /EHs-c-)
else()
  target_compile_options(viettelex_engine PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions -fno-rtti)
endif()

if(VTX_BUILD_TESTS)
  enable_testing()
  find_package(ZLIB)
  set(VTX_GOLDEN_GZ "${CMAKE_CURRENT_SOURCE_DIR}/../../android/telexcore/src/test/resources/golden.tsv.gz"
      CACHE FILEPATH "Golden corpus generated by `swift run gen-golden` in TelexCore/")

  add_executable(vtx_tests tests/test_main.cpp tests/golden_test.cpp tests/ported_tests.cpp
                           tests/unit_tests.cpp)
  target_link_libraries(vtx_tests PRIVATE viettelex_engine)
  target_include_directories(vtx_tests PRIVATE src tests)
  if(ZLIB_FOUND)
    target_link_libraries(vtx_tests PRIVATE ZLIB::ZLIB)
    target_compile_definitions(vtx_tests PRIVATE VTX_HAVE_ZLIB=1)
  endif()
  if(MSVC)
    target_compile_options(vtx_tests PRIVATE /utf-8)
  endif()
  add_test(NAME golden COMMAND vtx_tests golden "${VTX_GOLDEN_GZ}")
  add_test(NAME unit COMMAND vtx_tests unit)

  add_executable(vtx_bench bench/bench.cpp)
  target_link_libraries(vtx_bench PRIVATE viettelex_engine)
  add_executable(vtx_c_smoke tests/c_smoke.c)
  set_source_files_properties(tests/c_smoke.c PROPERTIES LANGUAGE C)
  enable_language(C)
  target_link_libraries(vtx_c_smoke PRIVATE viettelex_engine)
  if(NOT MSVC)
    set_target_properties(vtx_c_smoke PROPERTIES LINKER_LANGUAGE CXX)
  endif()
  add_test(NAME c_abi COMMAND vtx_c_smoke)
endif()
